## STEP 9 — Traces and observability (CloudWatch)                              [EC2]
## =====================================================================================
REGION=us-east-1
ACCOUNT=$(aws sts get-caller-identity --query Account --output text)

## Shortcut for 9.1 + 9.2 in one command:  uv run scripts/observability.py setup   (check: ... check)

## 9.1 Transaction Search (X-Ray spans -> CloudWatch Logs "aws/spans"). Once per account.
aws xray get-trace-segment-destination --region $REGION      # CloudWatchLogs + ACTIVE = already on, skip to 9.2
cat > /tmp/ts-policy.json <<EOF
{"Version":"2012-10-17","Statement":[{"Sid":"TransactionSearchXRayAccess","Effect":"Allow",
 "Principal":{"Service":"xray.amazonaws.com"},"Action":"logs:PutLogEvents",
 "Resource":["arn:aws:logs:$REGION:$ACCOUNT:log-group:aws/spans:*","arn:aws:logs:$REGION:$ACCOUNT:log-group:/aws/application-signals/data:*"],
 "Condition":{"ArnLike":{"aws:SourceArn":"arn:aws:xray:$REGION:$ACCOUNT:*"},"StringEquals":{"aws:SourceAccount":"$ACCOUNT"}}}]}
EOF
aws logs put-resource-policy --region $REGION --policy-name AgentCoreTransactionSearch --policy-document file:///tmp/ts-policy.json
aws xray update-trace-segment-destination --region $REGION --destination CloudWatchLogs
## Optional: index 100% of traces for search (above 1% is billed)
# aws xray update-indexing-rule --region $REGION --name Default --rule '{"Probabilistic":{"DesiredSamplingPercentage":100}}'

## 9.2 Gateway + Memory logs and traces (vended deliveries). Re-run after the memory is renamed/recreated.
GW_ID=$(O GatewayId); MEM_ID=$(O MemoryId); MEM_ARN=$(O MemoryArn)
GW_ARN=arn:aws:bedrock-agentcore:$REGION:$ACCOUNT:gateway/$GW_ID
for pair in "$GW_ID $GW_ARN" "$MEM_ID $MEM_ARN"; do
  set -- $pair; RID=$1; ARN=$2; GROUP=/aws/vendedlogs/bedrock-agentcore/$RID
  aws logs create-log-group --region $REGION --log-group-name $GROUP 2>/dev/null
  aws logs put-delivery-source --region $REGION --name $RID-logs-source --log-type APPLICATION_LOGS --resource-arn $ARN >/dev/null
  DST=$(aws logs put-delivery-destination --region $REGION --name $RID-logs-destination --delivery-destination-type CWL \
        --delivery-destination-configuration destinationResourceArn=arn:aws:logs:$REGION:$ACCOUNT:log-group:$GROUP \
        --query deliveryDestination.arn --output text)
  aws logs create-delivery --region $REGION --delivery-source-name $RID-logs-source --delivery-destination-arn $DST >/dev/null 2>&1
  aws logs put-delivery-source --region $REGION --name $RID-traces-source --log-type TRACES --resource-arn $ARN >/dev/null
  DST=$(aws logs put-delivery-destination --region $REGION --name $RID-traces-destination --delivery-destination-type XRAY \
        --query deliveryDestination.arn --output text)
  aws logs create-delivery --region $REGION --delivery-source-name $RID-traces-source --delivery-destination-arn $DST >/dev/null 2>&1
  echo "OK: logs + traces on for $RID"
done

## 9.3 Check deliveries exist (should list 4: gateway logs/traces, memory logs/traces)
aws logs describe-deliveries --region $REGION --query 'deliveries[].deliverySourceName' --output table

## 9.4 Generate traffic (STEP 7), wait ~10 minutes the first time, then find spans for a session
START=$(date -d '-1 hour' +%s); END=$(date +%s)
QID=$(aws logs start-query --region $REGION --log-group-name aws/spans --start-time $START --end-time $END \
      --query-string "fields @timestamp, name | filter @message like '$SID' | sort @timestamp asc | limit 100" \
      --query queryId --output text)
sleep 10; aws logs get-query-results --region $REGION --query-id $QID --output table

## 9.5 Gateway and Memory log groups
aws logs tail /aws/vendedlogs/bedrock-agentcore/$GW_ID --since 1h --region $REGION
aws logs tail /aws/vendedlogs/bedrock-agentcore/$MEM_ID --since 1h --region $REGION
